\begin{lstlisting}[language=C, basicstyle=\ttfamily]
void F(int a[], int i, int n)
{
    if(i >= n-1)
        return;
    // 不是冒泡排序,若要修改,则修改成注释循环部分
    // for(int j=0; j<n-1-i; j++)
    for(int j=i; j<n-1; j++)
    {
        if(a[j] > a[j+1])
        {
            int temp = a[j];
            a[j] = a[j+1];
            a[j+1] = temp;
        }
    }
    F(a, i+1, n);
}
